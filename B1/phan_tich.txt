def get_current_user(token: str = Depends(oauth2_scheme)):
    try:
        # LỖI 1: Đọc payload nhưng bỏ qua xác thực chữ ký và hạn dùng
        payload = jwt.get_unverified_claims(token)
    except Exception:
        raise HTTPException(...)

    username = payload.get("sub")
    user = USERS.get(username)

    if user is None:
        raise HTTPException(...)

    # LỖI 2: Chưa kiểm tra trạng thái is_active của user
    return user

# Nguyễn nhân get_unverified_claims () không an toàn : 
# Bỏ qua chữ Ký (Signature): Hàm này chỉ giải mã chuỗi Base64 của phần Payload mà không dùng SECRET_KEY để đối chiếu chữ ký. Kẻ gian có thẻ tùy ý sửa sub thành tên bất kỳ ai (ví dụ tài khoản admin) mà hệ thống vẫn chấp nhận 
# Bỏ qua hạn dùng (exp): Hàm không kiểm tra trường thời gian hết hạn , dẫn đến việc token đã hết hạn hạn nhiều ngày vẫn được coi là hợp lệ
